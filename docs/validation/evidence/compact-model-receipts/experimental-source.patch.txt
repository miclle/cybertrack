+#pragma once
+#include <cJSON.h>
+#include <cstring>
+#include <string>
+
+// Presentation only, after the control lock is released. Diagnostics and failures
+// retain the full snapshot. Every retained value comes from the same real reply.
+inline void CompactModelReceipt(std::string& reply) {
+    auto* full = cJSON_ParseWithOpts(reply.c_str(), nullptr, true);
+    const auto* operation = cJSON_GetObjectItemCaseSensitive(full, "operation");
+    const auto* result = cJSON_GetObjectItemCaseSensitive(full, "result");
+    const auto* fault = cJSON_GetObjectItemCaseSensitive(full, "fault");
+    bool action = false, draft = false;
+    if (cJSON_IsString(operation) && cJSON_IsString(result)) {
+        for (const char* name : {"move", "water", "bubbles", "ignition", "move_and_water", "sequence", "frenzy", "set_speed", "adjust_speed"})
+            if (!std::strcmp(operation->valuestring, name) && !std::strcmp(result->valuestring, "accepted")) action = true;
+        action |= !std::strcmp(operation->valuestring, "stop") && !std::strcmp(result->valuestring, "ok");
+        draft = !std::strcmp(operation->valuestring, "show_edit") &&
+            (!std::strcmp(result->valuestring, "draft_saved") || !std::strcmp(result->valuestring, "draft_cancelled"));
+    }
+    const bool normal = cJSON_IsString(fault) && (!std::strcmp(fault->valuestring, "none") ||
+        ((draft || (cJSON_IsString(operation) && !std::strcmp(operation->valuestring, "stop"))) && !std::strcmp(fault->valuestring, "stopped")));
+    if ((!action && !draft) || !normal) { cJSON_Delete(full); return; }
+    auto* compact = cJSON_CreateObject();
+    if (!compact) { cJSON_Delete(full); return; }
+    auto copy = [&](const char* key) {
+        const auto* item = cJSON_GetObjectItemCaseSensitive(full, key);
+        auto* clone = item ? cJSON_Duplicate(item, true) : nullptr;
+        if (!clone) return false;
+        if (!cJSON_AddItemToObject(compact, key, clone)) { cJSON_Delete(clone); return false; }
+        return true;
+    };
+    bool valid = true;
+    for (const char* key : {"operation", "result", "state", "fault", "physical_feedback", "boot_held", "logical_outputs", "starts_this_boot"})
+        if (!copy(key)) { valid = false; break; }
+    if (valid && draft) valid = copy("show_draft");
+    if (valid && action) {
+        for (const char* key : {"last_duration_ms", "remaining_ms", "move_last_duration_ms", "water_last_duration_ms",
+                               "bubbles_last_duration_ms", "ignition_last_duration_ms", "move_remaining_ms", "water_remaining_ms",
+                               "bubbles_remaining_ms", "ignition_remaining_ms", "move_direction", "move_speed_percent",
+                               "move_source", "water_source", "bubbles_source", "ignition_source", "sequence_active", "sequence_step",
+                               "sequence_steps", "sequence_result", "sequence_source", "frenzy_remaining_ms", "frenzy_last_duration_ms", "frenzy_source"})
+            if (!copy(key)) { valid = false; break; }
+    }
+    char* encoded = valid ? cJSON_PrintUnformatted(compact) : nullptr;
+    if (encoded) {
+        const auto size = std::strlen(encoded);
+        if (size < reply.size()) {
+            // Reuse the full reply's preallocated storage after acceptance.
+            std::memcpy(reply.data(), encoded, size);
+            reply.resize(size);
+        }
+        cJSON_free(encoded);
+    }
+    cJSON_Delete(compact);
+    cJSON_Delete(full);
+}

+def check(executable, playback):
+    trace=json.loads(gzip.decompress(CAPTURE.read_bytes()))
+    expected={row['id']:row for row in json.loads(PLAYBACK.read_bytes())['results']}

+            if 'name' in call:replies.append(firmware_replay(executable,row['initial_profile'],calls[:i+1],await_stt=True)[-2])
+        original=row.get('host_rpc_replies',[])
+        assert len(original)==len(replies)
+        full_bytes=compact_bytes=0;changes=0;receipts=[]
+        for old,new in zip(original,replies):

+                assert old==new;continue

+            full=json.loads(a);compact=json.loads(b)
