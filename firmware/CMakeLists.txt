cmake_minimum_required(VERSION 3.16)
project(cybertrack_host C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(cybertrack_control src/control.c)
target_include_directories(cybertrack_control PUBLIC include)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(cybertrack_control PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(cybertrack_json STATIC third_party/yyjson/yyjson.c)
target_include_directories(cybertrack_json SYSTEM PUBLIC third_party/yyjson)
target_compile_definitions(cybertrack_json PUBLIC
    YYJSON_DISABLE_WRITER=1 YYJSON_DISABLE_UTILS=1 YYJSON_DISABLE_NON_STANDARD=1)
add_library(cybertrack_protocol src/protocol.c src/replay_store.c src/freshness.c)
target_include_directories(cybertrack_protocol PUBLIC include)
target_link_libraries(cybertrack_protocol PUBLIC cybertrack_control PRIVATE cybertrack_json)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(cybertrack_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

include(CTest)
add_library(cybertrack_runtime src/runtime.c src/inputs.c src/inbox.c src/link.c)
target_include_directories(cybertrack_runtime PUBLIC include)
target_link_libraries(cybertrack_runtime PUBLIC cybertrack_protocol)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(cybertrack_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
